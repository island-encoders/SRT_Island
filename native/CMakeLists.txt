cmake_minimum_required(VERSION 3.16)
project(SrtIslandBond LANGUAGES C CXX)
set(ENABLE_BONDING ON CACHE BOOL "Enable native SRT socket groups" FORCE)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(ENABLE_APPS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
FetchContent_Declare(haivision_srt
    URL https://codeload.github.com/Haivision/srt/tar.gz/refs/tags/v1.5.4
    URL_HASH SHA256=d0a8b600fe1b4eaaf6277530e3cfc8f15b8ce4035f16af4a5eb5d4b123640cdd)
FetchContent_MakeAvailable(haivision_srt)
add_executable(srt-island-bond receiver.cpp)
target_compile_features(srt-island-bond PRIVATE cxx_std_17)
target_include_directories(srt-island-bond PRIVATE "${haivision_srt_SOURCE_DIR}/srtcore" "${haivision_srt_SOURCE_DIR}/common" "${haivision_srt_BINARY_DIR}")
target_link_libraries(srt-island-bond PRIVATE srt_static ws2_32)
if(MINGW)
    target_link_options(srt-island-bond PRIVATE -static-libgcc -static-libstdc++)
endif()
